<script>
  document.addEventListener('DOMContentLoaded', function () {
    console.log('DOM loaded');
    var btn = document.querySelector('.load-more-btn');
    var items = document.querySelectorAll('.loadmore-item');
    if (!btn || !items.length) return;
    items.forEach(function (item) { item.classList.remove('show'); });
    btn.addEventListener('click', function (e) {
      e.preventDefault();
      items.forEach(function (item) { item.classList.toggle('show'); });
      if (btn.innerHTML.includes('Load More')) {
        btn.innerHTML = 'Show Less <i class="fas fa-chevron-up ml-2"></i>';
      } else {
        btn.innerHTML = 'Load More <i class="fas fa-chevron-down ml-2"></i>';
      }
    });
  });
</script>
