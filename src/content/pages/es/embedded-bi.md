  .text-limit-3 { overflow: hidden; text-overflow: ellipsis; display: -webkit-box; -webkit-line-clamp: 3; -webkit-box-orient: vertical; }
  .text-limit-4 { overflow: hidden; text-overflow: ellipsis; display: -webkit-box; -webkit-line-clamp: 4; -webkit-box-orient: vertical; }
  @media (max-width: 767.98px) {
    .loadmore-item { opacity: 0; transform: translateY(20px); transition: opacity 0.5s ease, transform 0.5s ease; display: none !important; }
    .loadmore-item.show { opacity: 1; transform: translateY(0); display: flex !important; }
  }
</style>

<script>
  document.addEventListener('DOMContentLoaded', function () {
    console.log('DOM loaded');
    var btn = document.querySelector('.load-more-btn');
    var items = document.querySelectorAll('.loadmore-item');
    if (!btn || !items.length) return;
    items.forEach(function (item) { item.classList.remove('show'); });
    btn.addEventListener('click', function (e) {
      e.preventDefault();
      items.forEach(function (item) { item.classList.toggle('show'); });
      if (btn.innerHTML.includes('Load More')) {
        btn.innerHTML = 'Show Less <i class="fas fa-chevron-up ml-2"></i>';
      } else {
        btn.innerHTML = 'Load More <i class="fas fa-chevron-down ml-2"></i>';
      }
    });
  });
</script>
